# 前端页面示例代码

## 一、学生选课页面 (studentSelectCourse.vue)

```vue
<template>
  <div class="select-course-container">
    <el-card>
      <div slot="header">
        <span>课程选择</span>
      </div>
      
      <!-- 已选课程列表 -->
      <div class="selected-courses">
        <h3>已选课程</h3>
        <el-table :data="selectedCourses" style="width: 100%" v-if="selectedCourses.length > 0">
          <el-table-column prop="courseName" label="课程名称" width="180"></el-table-column>
          <el-table-column prop="courseCode" label="课程代码" width="120"></el-table-column>
          <el-table-column prop="credits" label="学分" width="80"></el-table-column>
          <el-table-column prop="institute" label="所属学院" width="150"></el-table-column>
          <el-table-column label="操作" width="120">
            <template slot-scope="scope">
              <el-button size="mini" type="danger" @click="dropCourse(scope.row.courseId)">退课</el-button>
            </template>
          </el-table-column>
        </el-table>
        <el-empty v-else description="暂无已选课程"></el-empty>
      </div>

      <el-divider></el-divider>

      <!-- 可选课程列表 -->
      <div class="available-courses">
        <h3>可选课程</h3>
        <el-table :data="availableCourses" style="width: 100%">
          <el-table-column prop="courseName" label="课程名称" width="180"></el-table-column>
          <el-table-column prop="courseCode" label="课程代码" width="120"></el-table-column>
          <el-table-column prop="credits" label="学分" width="80"></el-table-column>
          <el-table-column prop="institute" label="所属学院" width="150"></el-table-column>
          <el-table-column prop="description" label="课程描述" show-overflow-tooltip></el-table-column>
          <el-table-column label="选择教师" width="200">
            <template slot-scope="scope">
              <el-select v-model="selectedTeachers[scope.row.courseId]" 
                         placeholder="请选择教师" 
                         @change="handleTeacherChange(scope.row.courseId, $event)"
                         size="mini">
                <el-option
                  v-for="teacher in courseTeachers[scope.row.courseId]"
                  :key="teacher.teacherId"
                  :label="teacher.teacherName + ' (' + teacher.institute + ')'"
                  :value="teacher.teacherId">
                </el-option>
              </el-select>
            </template>
          </el-table-column>
          <el-table-column label="操作" width="120">
            <template slot-scope="scope">
              <el-button 
                size="mini" 
                type="primary" 
                @click="selectCourse(scope.row)"
                :disabled="!selectedTeachers[scope.row.courseId]">
                选课
              </el-button>
            </template>
          </el-table-column>
        </el-table>
      </div>
    </el-card>
  </div>
</template>

<script>
export default {
  name: 'StudentSelectCourse',
  data() {
    return {
      allCourses: [],
      selectedCourses: [],
      availableCourses: [],
      courseTeachers: {}, // {courseId: [teachers]}
      selectedTeachers: {} // {courseId: teacherId}
    }
  },
  mounted() {
    this.loadSelectedCourses();
    this.loadAllCourses();
  },
  methods: {
    // 加载已选课程
    async loadSelectedCourses() {
      try {
        const res = await this.$axios.get('/student/courses');
        if (res.data.code === 200) {
          this.selectedCourses = res.data.data || [];
        }
      } catch (error) {
        this.$message.error('加载已选课程失败');
      }
    },
    
    // 加载所有课程
    async loadAllCourses() {
      try {
        const res = await this.$axios.get('/courses');
        if (res.data.code === 200) {
          this.allCourses = res.data.data || [];
          this.filterAvailableCourses();
          
          // 为每个课程加载教师列表
          for (let course of this.allCourses) {
            await this.loadTeachersForCourse(course.courseId);
          }
        }
      } catch (error) {
        this.$message.error('加载课程列表失败');
      }
    },
    
    // 过滤可选课程（排除已选课程）
    filterAvailableCourses() {
      const selectedCourseIds = this.selectedCourses.map(c => c.courseId);
      this.availableCourses = this.allCourses.filter(
        course => !selectedCourseIds.includes(course.courseId)
      );
    },
    
    // 加载某个课程的教师列表
    async loadTeachersForCourse(courseId) {
      try {
        const res = await this.$axios.get(`/course/${courseId}/teachers`);
        if (res.data.code === 200) {
          this.$set(this.courseTeachers, courseId, res.data.data || []);
        }
      } catch (error) {
        console.error(`加载课程 ${courseId} 的教师列表失败`, error);
      }
    },
    
    // 教师选择变化
    handleTeacherChange(courseId, teacherId) {
      this.$set(this.selectedTeachers, courseId, teacherId);
    },
    
    // 选课
    async selectCourse(course) {
      const teacherId = this.selectedTeachers[course.courseId];
      if (!teacherId) {
        this.$message.warning('请先选择教师');
        return;
      }
      
      try {
        const res = await this.$axios.post('/student/course', {
          courseId: course.courseId,
          teacherId: teacherId
        });
        
        if (res.data.code === 200) {
          this.$message.success('选课成功');
          // 重新加载数据
          await this.loadSelectedCourses();
          this.filterAvailableCourses();
          // 清除选择
          this.$delete(this.selectedTeachers, course.courseId);
        } else {
          this.$message.error(res.data.message || '选课失败');
        }
      } catch (error) {
        this.$message.error('选课失败：' + (error.response?.data?.message || error.message));
      }
    },
    
    // 退课
    async dropCourse(courseId) {
      try {
        await this.$confirm('确定要退选该课程吗？', '提示', {
          confirmButtonText: '确定',
          cancelButtonText: '取消',
          type: 'warning'
        });
        
        const res = await this.$axios.delete(`/student/course/${courseId}`);
        if (res.data.code === 200) {
          this.$message.success('退课成功');
          // 重新加载数据
          await this.loadSelectedCourses();
          this.filterAvailableCourses();
        } else {
          this.$message.error(res.data.message || '退课失败');
        }
      } catch (error) {
        if (error !== 'cancel') {
          this.$message.error('退课失败：' + (error.response?.data?.message || error.message));
        }
      }
    }
  }
}
</script>

<style scoped>
.select-course-container {
  padding: 20px;
}

.selected-courses, .available-courses {
  margin-bottom: 20px;
}

h3 {
  margin-bottom: 15px;
  color: #303133;
}
</style>
```

## 二、教师发布试卷页面修改 (addExam.vue)

需要在现有的发布试卷表单中添加课程选择字段：

```vue
<template>
  <!-- 在现有的表单中添加课程选择 -->
  <el-form-item label="所属课程" prop="courseId" required>
    <el-select 
      v-model="examForm.courseId" 
      placeholder="请选择课程"
      @change="handleCourseChange"
      style="width: 100%">
      <el-option
        v-for="course in teacherCourses"
        :key="course.courseId"
        :label="course.courseName + ' (' + course.courseCode + ')'"
        :value="course.courseId">
      </el-option>
    </el-select>
    <div style="color: #909399; font-size: 12px; margin-top: 5px;">
      只能选择您教授的课程
    </div>
  </el-form-item>
</template>

<script>
export default {
  data() {
    return {
      examForm: {
        // ... 其他字段
        courseId: null
      },
      teacherCourses: [] // 教师教授的课程列表
    }
  },
  mounted() {
    this.loadTeacherCourses();
  },
  methods: {
    // 加载教师教授的课程列表
    async loadTeacherCourses() {
      try {
        const res = await this.$axios.get('/teacher/courses');
        if (res.data.code === 200) {
          this.teacherCourses = res.data.data || [];
        }
      } catch (error) {
        this.$message.error('加载课程列表失败');
      }
    },
    
    // 课程选择变化时，自动填充课程名称到source字段
    handleCourseChange(courseId) {
      const course = this.teacherCourses.find(c => c.courseId === courseId);
      if (course) {
        this.examForm.source = course.courseName;
      }
    },
    
    // 提交表单时验证
    async submitForm() {
      if (!this.examForm.courseId) {
        this.$message.warning('请选择所属课程');
        return;
      }
      
      // ... 其他提交逻辑
      const res = await this.$axios.post('/exam', this.examForm);
      // ...
    }
  }
}
</script>
```

## 四、API接口说明

### 学生选课相关
- `GET /student/courses` - 获取学生已选的课程列表
- `POST /student/course` - 学生选课（需要传递 courseId 和 teacherId）
- `DELETE /student/course/{courseId}` - 学生退课
- `GET /courses` - 获取所有课程列表
- `GET /course/{courseId}/teachers` - 获取某个课程的教师列表

### 教师相关
- `GET /teacher/courses` - 获取教师教授的课程列表
- `POST /exam` - 发布试卷（需要传递 courseId）

## 五、使用说明

1. **学生选课流程**：
   - 学生登录后，进入选课页面
   - 查看可选课程列表
   - 为每个课程选择教师
   - 点击"选课"按钮完成选课
   - 可以在"已选课程"中查看和管理已选课程

2. **教师发布试卷流程**：
   - 教师登录后，进入发布试卷页面
   - 在"所属课程"下拉框中选择课程（只能选择自己教授的课程）
   - 填写其他试卷信息
   - 提交后，只有选择了该课程的学生才能看到该试卷
